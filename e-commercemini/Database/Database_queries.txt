create database ecommerce;
use ecommerce;
CREATE TABLE users(
id INT AUTO_INCREMENT PRIMARY KEY,
name VARCHAR(50),
email VARCHAR(50),
password VARCHAR(50)
);
CREATE TABLE products(
id INT AUTO_INCREMENT PRIMARY KEY,
name VARCHAR(100),
price DOUBLE,
description VARCHAR(200),
image VARCHAR(200)
);
CREATE TABLE cart(
id INT AUTO_INCREMENT PRIMARY KEY,
product_id INT,
quantity INT
);
